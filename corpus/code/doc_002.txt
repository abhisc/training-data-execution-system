import hashlib
from pathlib import Path

def content_hash(path: Path) -> str:
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(65536), b""):
            h.update(chunk)
    return h.hexdigest()

def pack_tokens(seqs, max_len):
    out = []
    for seq in seqs:
        out.extend(seq[:max_len])
    return out
